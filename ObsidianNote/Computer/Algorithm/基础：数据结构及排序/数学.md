```cpp
bool st[N];           // 标记数组，st[i] == true 表示 i 被筛掉（不是质数）
int primes[N], cnt;   // primes数组存储筛出来的质数，cnt是质数数量

void getprimes(int n) {
    st[0] = st[1] = true;   // 0 和 1 不是质数，标记为true（筛掉）
    for (int i = 2; i <= n; i++) {
        if (!st[i]) {       // 如果 i 没被筛掉，说明 i 是质数
            primes[cnt++] = i;    // 把质数 i 存入primes数组
            // 将 i 的所有倍数标记为非质数（筛掉）
            for (int j = i + i; j <= n; j += i) {
                st[j] = true;
            }
        }
    }
}

```

```cpp
class Solution {

    int distinctPrimeFactors(vector<int>& nums) {
        unordered_set<int> primes;
        for (int& num: nums) {
            // 寻找 num 的质因数
            int ans = 0;           
            for (int i = 2; i * i <= num; ++i) {

                while ((num % i) == 0) {                    
                    num /= i;
                }                               
            }
            // 循环结束后，添加剩余的质因数 
            if (num > 1) primes.insert(num);  
        }
        return primes.size();
    }
```

## 最小公倍数

根据数学定理：

$\text{GCD}(a, b) \times \text{LCM}(a, b) = a \times b$

所以：

$\text{LCM}(a, b) = \frac{a \times b}{\text{GCD}(a, b)}$

```cpp
int lcm(int a, int b) {
    return a / gcd(a, b) * b;  // 先除后乘避免溢出
}
```

## 快速幂

快速幂（又称二分幂）是计算$a^b mod  m$的高效方法，避免了直接暴力乘法的高时间复杂度。

算法原理

- 利用指数的二进制表示，将幂次拆分成若干幂的乘积；
- 递归/迭代地计算 $a^{2^k}$ 的结果，通过不断平方实现；
- 只对二进制位为1的幂进行乘积累乘。

```cpp
long long mod_pow(long long a, long long b, long long m) {
    long long result = 1 % m; // 初始化为1
    long long cur = a % m; // 当前幂次的值

    while (b > 0) {
        if (b & 1) {
            result = (result * cur) % m; // 当前位为1，乘入结果
        }
        cur = (cur * cur) % m; // 平方
        b >>= 1;               // 右移，处理下一位
    }

    return result;
}

```

## 唯一分解定理

**定理内容：**

任何大于1的整数，都可以写成若干个素数的乘积，而且这种分解在素数的选择及其幂次上是唯一的（忽略乘积中素数的排列顺序）。

换句话说，任意整数 n>1n > 1n>1 都有且只有一种方式表示为：

$n= p_1^{\alpha_1} \times p_2^{\alpha_2} \times \cdots \times p_k^{\alpha_k}$

其中：

- $p_1, p_2, \ldots, p_k$ 是互不相同的素数（质数）；
- $\alpha_1, \alpha_2, \ldots, \alpha_k$ 是正整数。

```cpp
#include <iostream>
using namespace std;

void prime_factorization(int n) {
    cout << n << "=";
    bool first = true; // 控制输出格式，避免最后多余 '*'

    // 从2开始尝试除数
    for (int i = 2; i * i <= n; i++) {
        while (n % i == 0) {
            if (!first) cout << "*";
            cout << i;
            first = false;
            n /= i;
        }
    }
    // 如果剩余大于1，则是最后一个素因数
    if (n > 1) {
        if (!first) cout << "*";
        cout << n;
    }
    cout << endl;
}

int main() {
    int a, b;
    cin >> a >> b;
    for (int i = a; i <= b; i++) {
        prime_factorization(i);
    }
    return 0;
}

```

## 阶乘后的尾随零

原理说明：

- 阶乘末尾的零是由因子 10 产生的，10 = 2 × 5；
- 在阶乘中，因子2的数量远多于因子5，所以决定末尾零的主要是因子5的数量；
- 因此，计算阶乘中因子5的个数即为末尾零的个数。

```cpp
int trailingZeroes(int n) {
    int ans = 0;
    while (n) {
        n /= 5;       // 统计因子5的个数
        ans += n;     // 累加
    }
    return ans;
}

```

## 逆元

### 扩展欧几里得

```cpp
#include <bits/stdc++.h>
using namespace std;

// 扩展欧几里得算法
// 返回gcd(a,b)，并通过引用x,y返回方程ax + by = gcd(a,b)的解
int extended_gcd(int a, int b, int &x, int &y) {
    if (b == 0) { 
        x = 1; y = 0; 
        return a;
    }
    int x1, y1;
    int gcd = extended_gcd(b, a % b, x1, y1);
    x = y1;
    y = x1 - (a / b) * y1;
    return gcd;
}

// 计算a模m的逆元，假设a和m互质
int mod_inverse(int a, int m) {
    int x, y;
    int gcd = extended_gcd(a, m, x, y);
    if (gcd != 1) {
        // 不互质无逆元
        return -1;
    } else {
        // 保证逆元是正数
        return (x % m + m) % m;
    }
}

int main() {
    int a, m;
    cin >> a >> m;
    int inv = mod_inverse(a, m);
    if (inv == -1)
        cout << "No inverse exists." << endl;
    else
        cout << inv << endl;
    return 0;
}

```

### 快速幂（求质数逆元）

适合质数模数 p 下的逆元计算，利用费马小定理：

$a^{-1} \equiv a^{p-2} \pmod{p}$

```cpp
#include <iostream>
using namespace std;

const int MOD = 1000000007;

// 快速幂函数：计算 a^b mod MOD
long long mod_pow(long long a, long long b, int mod = MOD) {
    long long res = 1;
    a %= mod;
    while (b > 0) {
        if (b & 1) res = (res * a) % mod;  // b的当前位为1，则乘上a的当前幂
        a = (a * a) % mod;                  // a平方，幂数减半
        b >>= 1;
    }
    return res;
}

// 计算模逆元，a^(MOD-2) mod MOD
long long mod_inv(long long a, int mod = MOD) {
    return mod_pow(a, mod - 2, mod);
}

int main() {
    long long n;
    cin >> n;

    cout << mod_inv(n) << "\\n";

    return 0;
}

```

## 高精度计算

### 1. 高精度加法

**思路**：模拟小学的加法，从低位向高位逐位相加，处理进位。

**关键点**：

- 数字通常逆序存储（低位在前）便于从个位开始加。
- 按位相加，最后注意最高位可能产生的进位。

```cpp
vector<int> add(const vector<int>& a, const vector<int>& b) {
    vector<int> res;
    int carry = 0;
    int n = max(a.size(), b.size());
    for (int i = 0; i < n; i++) {
        int x = i < (int)a.size() ? a[i] : 0;
        int y = i < (int)b.size() ? b[i] : 0;
        int sum = x + y + carry;
        res.push_back(sum % 10);
        carry = sum / 10;
    }
    if (carry) res.push_back(carry);
    return res;
}

```

### 2. 高精度减法

**思路**：模拟小学的减法，从低位向高位逐位减，处理借位。通常约定第一个数≥第二个数。

**关键点**：

- 同样逆序存储。
- 减时若被减位小于减位数，需向高位借1。
- 结果去除高位多余的0。

```cpp
vector<int> sub(const vector<int>& a, const vector<int>& b) {
    vector<int> res = a;
    for (int i = 0; i < (int)b.size(); i++) {
        res[i] -= b[i];
        if (res[i] < 0) {
            res[i] += 10;
            res[i+1]--;
        }
    }
    for (int i = (int)b.size(); i < (int)a.size(); i++) {
        if (res[i] < 0) {
            res[i] += 10;
            res[i+1]--;
        }
    }
    // 去除高位0
    while (res.size() > 1 && res.back() == 0) res.pop_back();
    return res;
}

```

### 3. 高精度乘法

**思路**：模拟小学乘法，逐位相乘后累加到对应位置，处理进位。

**关键点**：

- 结果最大长度为 `a.size()` + `b.size()`。
- 逐位相乘后累加进结果数组。
- 最后处理所有进位，去除高位0。

```cpp
vector<int> multiply(const vector<int>& a, const vector<int>& b) {
    vector<int> res(a.size() + b.size(), 0);
    for (int i = 0; i < (int)a.size(); i++) {
        int carry = 0;
        for (int j = 0; j < (int)b.size(); j++) {
            int tmp = res[i+j] + a[i]*b[j] + carry;
            res[i+j] = tmp % 10;
            carry = tmp / 10;
        }
        if (carry) res[i + (int)b.size()] += carry;
    }
    while (res.size() > 1 && res.back() == 0) res.pop_back();
    return res;
}

```

### 4. 高精度除法

**思路**：模拟小学长除法，从高位开始尝试除，逐步获得商和余数。

**关键点**：

- 除数为整数或者高精度数都可以，后者实现较复杂。
- 逐步“拉下”被除数高位，判断商的当前位。
- 注意商的高位可能是0，去除高位0。

```cpp
// 这里只示意高精度除以整数的情况，返回商和余数
pair<vector<int>, int> divide(const vector<int>& a, int b) {
    vector<int> res;
    int remainder = 0;
    for (int i = (int)a.size()-1; i >= 0; i--) {
        remainder = remainder * 10 + a[i];
        res.push_back(remainder / b);
        remainder %= b;
    }
    reverse(res.begin(), res.end());
    // 去除高位0
    while (res.size() > 1 && res.back() == 0) res.pop_back();
    return {res, remainder};
}

```

### 矩阵乘法

```cpp
void multiplyMatrix(int A[][N], int B[][N], int C[][N], int m, int n, int p) {
    // A: m×n矩阵, B: n×p矩阵, C: m×p矩阵
    for(int i = 0; i < m; i++) {
        for(int j = 0; j < p; j++) {
            C[i][j] = 0;
            for(int k = 0; k < n; k++) {
                C[i][j] += A[i][k] * B[k][j];
            }
        }
    }
}
```

## 闰年判断 + 每月天数表

闰年：年份能够**被4整除但不能被100整除**或者**能被400整除**的为闰年。

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    int year, month;
    cin >> year >> month;

    // 每个月的天数（默认平年）
    int days_in_month[] = {0, 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};

    // 判断闰年
    bool leap = ( (year % 4 == 0 && year % 100 != 0) || (year % 400 == 0) );
    if (leap) days_in_month[2] = 29; // 闰年 2 月变 29 天

    cout << days_in_month[month] << endl;
    return 0;
}

```

### 约瑟夫环
```cpp
#include <iostream>

// 返回在n个人中，每数到m时杀掉一个，最后剩下的人的编号（从1开始）
int josephus(int n, int m) {
    int res = 0; // 当只有1个人时，胜利者编号是0（0-based）
    for (int i = 2; i <= n; i++) {
        res = (res + m) % i;
    }
    return res + 1; // 转换成1-based编号
}
```
